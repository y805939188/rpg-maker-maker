def png_pixels(path):

 while offset<len(b):
  size=struct.unpack_from('>I',b,offset)[0];tag=b[offset+4:offset+8];data=b[offset+8:offset+8+size];crc=struct.unpack_from('>I',b,offset+8+size)[0];assert zlib.crc32(tag+data)&0xffffffff==crc
  if tag==b'IHDR':w,h,depth,kind,compression,filtering,interlace=struct.unpack('>IIBBBBB',data);assert (depth,kind,compression,filtering,interlace)==(8,6,0,0,0)
  if tag==b'IDAT':idat+=data
  chunks.append(tag.decode());offset+=size+12
 assert chunks[-1]=='IEND' and offset==len(b)
 raw=zlib.decompress(idat);stride=w*4;assert len(raw)==h*(stride+1);pixels=bytearray();previous=bytearray(stride)
 for y in range(h):
  method=raw[y*(stride+1)];row=bytearray(raw[y*(stride+1)+1:(y+1)*(stride+1)])
  for x in range(stride):
   left=row[x-4] if x>=4 else 0;up=previous[x];corner=previous[x-4] if x>=4 else 0
   if method==0:p=0
   elif method==1:p=left
   elif method==2:p=up
   elif method==3:p=(left+up)//2
   else:
    assert method==4;pred=left+up-corner;dist=[abs(pred-v) for v in (left,up,corner)];p=(left,up,corner)[dist.index(min(dist))]
   row[x]=(row[x]+p)&255
  pixels.extend(row);previous=row
 return {'width':w,'height':h,'chunks':chunks,'rgba':bytes(pixels)}
