import datetime,hashlib,json,os,pathlib,subprocess,sys
root=pathlib.Path(__file__).resolve().parents[2]
out=root/'.local/review'
label,*command=sys.argv[1:]
env=os.environ.copy()
env['PATH']='/Users/dingshinn/.nvm/versions/node/v22.21.1/bin:'+env['PATH']
env['XDG_CACHE_HOME']=str(root/'.local/cache')
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
with (out/(label+'.log')).open('w') as log:
 result=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
record={'label':label,'command':command,'commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),'start':start,'end':datetime.datetime.now(datetime.timezone.utc).isoformat(),'exit':result.returncode,'logSha256':hashlib.sha256((out/(label+'.log')).read_bytes()).hexdigest()}
with (out/'commands.jsonl').open('a') as f:f.write(json.dumps(record)+'\n')
print(json.dumps(record))
print((out/(label+'.log')).read_text()[-2800:])
sys.exit(result.returncode)
