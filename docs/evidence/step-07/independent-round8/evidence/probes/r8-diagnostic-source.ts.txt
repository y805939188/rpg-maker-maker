  const t=performance.now();
  const response = await r8NativeFetch(...args);
  const actualUrl=typeof args[0]==='string'?args[0]:args[0] instanceof URL?args[0].href:args[0].url;
  if (/\/processed\?frame=/.test(actualUrl)) {

    await writeFile(resolve(r8CaptureRoot,'processed-calls.json'),JSON.stringify(r8ProcessedCalls,null,2)+'\n');
  }
  if (response.status!==200 || !actualUrl.endsWith('/download')) return response;
  const index=r8DownloadIndex++, original=Buffer.from(await response.clone().arrayBuffer());
  await writeFile(resolve(r8CaptureRoot,`download-${index}-original.zip`),original);
  if(r8Variant!==`fault-${index}`)return response;
  const mutated=Buffer.from(original), end=mutated.length-22, central=mutated.readUInt32LE(end+16);
  mutated[central+16]!^=1;
  const headers=new Headers(response.headers);headers.set('x-content-sha256',hashBytes(mutated));
  await writeFile(resolve(r8CaptureRoot,`download-${index}-central-crc.zip`),mutated);
  await writeFile(resolve(r8CaptureRoot,'mutation.json'),JSON.stringify({index,url:actualUrl,status:response.status,originalSHA:hashBytes(original),badSHA:hashBytes(mutated),centralCRCBitChanged:true,offset:central+16,outerBodySHARecomputed:headers.get('x-content-sha256'),exportManifestHeader:headers.get('x-export-manifest-sha256')},null,2)+'\n');
  return new Response(mutated,{status:response.status,statusText:response.statusText,headers});
};

async function r(path: string, body?: unknown) {

function crc(bytes: Uint8Array) {
  let c = 0xffffffff;
  for (const v of bytes) {
    c ^= v;
    for (let n = 0; n < 8; n++) c = (c >>> 1) ^ (c & 1 ? 0xedb88320 : 0);
  }
  return (c ^ 0xffffffff) >>> 0;
}
function chunk(name: string, payload: Uint8Array) {
  const body = Buffer.concat([Buffer.from(name), Buffer.from(payload)]),
    out = Buffer.alloc(body.length + 8);
  out.writeUInt32BE(payload.length);
  body.copy(out, 4);
  out.writeUInt32BE(crc(body), out.length - 4);
  return out;
}
function pngWire(width = 96, height = 144, type = 6, meta?: string, hidden = false) {
  const h = Buffer.alloc(13);
  h.writeUInt32BE(width);
  h.writeUInt32BE(height, 4);
  h[8] = 8;
  h[9] = type;
  const rows: Buffer[] = [];
  let random = 123;
  for (let y = 0; y < height; y++) {
    const row = Buffer.alloc(1 + width * 4);
    for (let x = 0; x < width; x++) {
      random = (random * 1664525 + 1013904223) >>> 0;
      row.set(
        [random & 255, (random >>> 8) & 255, (random >>> 16) & 255, hidden ? 0 : 255],
        1 + x * 4,
      );
    }
    rows.push(row);
  }
  return Buffer.concat([
    Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]),
    chunk('IHDR', h),
    ...(meta
      ? [
          chunk(
            meta,
            meta === 'acTL'
              ? new Uint8Array([0, 0, 0, 1, 0, 0, 0, 0])
              : new Uint8Array([97, 0, 98]),
          ),
        ]
      : []),
    chunk('IDAT', deflateSync(Buffer.concat(rows), { level: 0 })),
    chunk('IEND', new Uint8Array()),
  ]);
}

          mode: 'update',
        }),
        prepared = await made;
      const bundle = JSON.parse(
          new TextDecoder().decode(prepared.files['bundle.json']!),
        ) as Parameters<typeof assembleExport>[1],
        standard = await assembleExport(resources.profile, bundle, resources.context);
      const failing = scenario === 'file-failure' || scenario === 'db-rollback';
      if (failing) {
        const snap = await external.projectSnapshot(f.project);
        await external.adopt(f.project, f.revision, RefSchema.array().parse(snap.adoption));
      }
      await external.publish(
        prepared,
        scenario === 'existing-standard' ? standard.files : undefined,
      );
      const before = await service.store.db.project.findUniqueOrThrow({ where: { id: f.project } }),
        oldInternal = hashBytes(
          (await service.store.storage.buildFiles(prepared.id))['manifest.json']!,
        );
      if (scenario === 'file-failure') r4PublicationFault = 'file-partial';
      let commitEntered: Promise<void> | undefined;
      if (scenario === 'db-rollback') {
        r4PublicationFault = 'db-transaction';
        let entered!: () => void;
        commitEntered = new Promise<void>((resolve) => {
          entered = resolve;
        });
        const barrier = new Promise<void>((resolve) => {
          allowCommit = resolve;
        });
        r4CommitFence = async () => {
          entered();
          await barrier;
        };
      }
      if (scenario === 'lost-receipt') r4PublicationFault = 'after-db-commit';
      release();
      if (commitEntered) {
        await commitEntered;
        // Actual PG transaction has inserted refs/completion internally, but no
        // public reader may observe them before COMMIT; then force rollback.
        expect(await service.store.db.finalExport.count({ where: { buildId: prepared.id } })).toBe(
          0,
        );
        expect(
          (await service.store.db.studioJob.findUniqueOrThrow({ where: { id: accepted.id } }))
            .state,
        ).toBe('RUNNING');
        expect(
          await service.store.db.project.findUniqueOrThrow({ where: { id: f.project } }),
        ).toEqual(before);
        allowCommit!();
      }
      const job = await r4Terminal(f.project, accepted.id),
        count = await service.store.db.finalExport.count({ where: { buildId: prepared.id } });
      await record('r4-finalization-' + scenario, {
        sourceContext: resources.context,
        scenario,
        job,
        count,
        before,
        after: await service.store.db.project.findUniqueOrThrow({ where: { id: f.project } }),
        internalManifest: oldInternal,
      });
      expect(job.state).toBe(failing ? 'FAILED' : 'COMPLETED');
      expect(count).toBe(failing ? 0 : 1);
      expect(
        await service.store.db.project.findUniqueOrThrow({ where: { id: f.project } }),
      ).toEqual(before);
      expect(
        hashBytes((await service.store.storage.buildFiles(prepared.id))['manifest.json']!),
      ).toBe(oldInternal);
      if (failing) {
        expect(before.successfulBuildId).toBe(normal.buildId);
        expect(
          hashBytes(
            new Uint8Array(
              await (
                await fetch(`${url}/api/projects/${f.project}/builds/${normal.buildId}/download`)
              ).arrayBuffer(),
            ),
          ),
        ).toBe(firstSHA);
        // Existing immutable files are retained; explicit local append/retry never changes failed Job to a fake success.
        await service.store.publish(prepared, standard.files);
        expect(
          (await service.store.db.studioJob.findUniqueOrThrow({ where: { id: accepted.id } }))
            .state,
        ).toBe('FAILED');
        expect(await service.store.db.finalExport.count({ where: { buildId: prepared.id } })).toBe(
          1,
        );
        expect(
          (await fetch(`${url}/api/projects/${f.project}/builds/${prepared.id}/download`)).status,
        ).toBe(200);
        service.store.prepareBuild = prepare;
        const fresh = await r(`/api/projects/${f.project}/studio`),
          retry = await r(`/api/projects/${f.project}/builds`, {
            recipeId: f.recipeId,
            expectedRevision: fresh.revision,
            mode: 'update',
          });
        expect((await r4Terminal(f.project, retry.id)).state).toBe('COMPLETED');
      } else {
        const files = await service.store.readExport(prepared.id);
        await service.store.publish(prepared, undefined, accepted.id);
        await service.store.publish(prepared, files.files, accepted.id);
        expect(await service.store.db.finalExport.count({ where: { buildId: prepared.id } })).toBe(
          1,
        );
        expect(
          (await fetch(`${url}/api/projects/${f.project}/builds/${prepared.id}/download`)).status,
        ).toBe(200);
      }
    } finally {
      allowCommit?.();
      r4CommitFence = null;
      release?.();
      r4PublicationFault = null;
      service.store.prepareBuild = prepare;
      await external.close();
    }
  },
);

    pending = stage;
    const start = performance.now();
    process.stderr.write(
      'R4_STAGE ' +
        JSON.stringify({ stage, event: 'start', elapsedMs: Math.round(start - started) }) +
        '\n',
    );
    await record('r4-ui-progress', { stage, elapsedMs: Math.round(start - started), timing });
    try {
      const value = await fn();
      timing.push({ stage, ms: Math.round(performance.now() - start), status: 'completed' });
      return value;
    } catch (error) {
      timing.push({ stage, ms: Math.round(performance.now() - start), status: 'failed' });
      throw error;
    } finally {
      process.stderr.write(
        'R4_STAGE ' +
          JSON.stringify({ stage, event: 'end', ms: Math.round(performance.now() - start) }) +
          '\n',
      );
      await record('r4-ui-progress', {
        stage,
        elapsedMs: Math.round(performance.now() - started),
        timing,
      });
    }
  }
