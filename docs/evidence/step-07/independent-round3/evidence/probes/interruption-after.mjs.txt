import {chromium,expect} from '@playwright/test';
import {readFile,writeFile} from 'node:fs/promises';
import {createHash} from 'node:crypto';
const record=JSON.parse(await readFile('.local/step07-review/evidence/interruption-before.json','utf8')),base='http://127.0.0.1:4383',s=await(await fetch(`${base}/api/projects/${record.project}/studio`)).json(),j=s.jobs.find(j=>j.id===record.acceptedJob.id);expect(j.state).toBe('FAILED');expect(j.diagnostic.code).toBe('INTERRUPTED');expect(s.successfulBuildId).toBe(record.successfulBefore);
const browser=await chromium.launch({headless:true}),page=await browser.newPage({viewport:{width:1280,height:960},acceptDownloads:true});const errors=[];page.on('pageerror',e=>errors.push(e.message));await page.goto(base);const projects=(await(await fetch(base+'/api/projects')).json()).projects;await page.getByRole('button',{name:projects.find(p=>p.id===record.project).name,exact:true}).click();await expect(page.getByRole('alert').filter({hasText:'上次本地构建中断'})).toContainText('上次本地构建中断');await expect(page.locator('canvas[aria-label="交付拼接预览"]')).toHaveAttribute('data-build-id',record.successfulBefore,{timeout:45000});await page.screenshot({path:'.local/step07-review/evidence/restored-browser/05-interrupted-recovered.png',fullPage:true});const wait=page.waitForEvent('download');await page.getByRole('link',{name:'下载此构建的标准 MZ 素材包 ZIP'}).click();await(await wait).saveAs('.local/step07-review/evidence/restored-browser/after-interruption.zip');const before=await readFile('.local/step07-review/evidence/restored-browser/before-interruption.zip'),after=await readFile('.local/step07-review/evidence/restored-browser/after-interruption.zip');expect(after.equals(before)).toBe(true);expect(errors).toEqual([]);await writeFile('.local/step07-review/evidence/interruption-after.json',JSON.stringify({job:j,lastSuccess:s.successfulBuildId,zipUnchanged:true,zipSha:createHash('sha256').update(after).digest('hex'),pageErrors:errors},null,2)+'\n');await browser.close();console.log('PASS actual SIGKILL/restart interrupted reason and last-success canvas/ZIP unchanged');
