import {chromium,expect} from '@playwright/test';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
const rec=JSON.parse(await readFile('.local/step07-review/evidence/integrated-recovery.json','utf8')),base='http://127.0.0.1:4383',out='.local/step07-review/evidence/poll-latency';await mkdir(out,{recursive:true});
const browser=await chromium.launch({headless:true}),context=await browser.newContext({viewport:{width:1280,height:960}});await context.tracing.start({screenshots:true,snapshots:true,sources:true});const page=await context.newPage();
const log={delayMs:1300,requests:[],responses:[],delayedPayloads:[],console:[],pageErrors:[]};
page.on('request',r=>{if(r.url().includes('/api/'))log.requests.push({url:r.url(),method:r.method(),time:Date.now()})});page.on('response',r=>{if(r.url().includes('/api/'))log.responses.push({url:r.url(),status:r.status(),time:Date.now()})});page.on('console',m=>log.console.push({type:m.type(),text:m.text()}));page.on('pageerror',e=>log.pageErrors.push(e.message));
try{
 const project=(await(await fetch(base+'/api/projects')).json()).projects.find(p=>p.id===rec.browserProject);await page.goto(base);await page.getByRole('button',{name:project.name,exact:true}).click();await expect(page.getByRole('heading',{name:project.name+' · 创作素材'})).toBeVisible();await expect(page.getByRole('button',{name:'构建当前配套',exact:true})).toBeEnabled();await expect(page.getByText('画布已就绪 · 实际导出图集',{exact:false})).toBeVisible({timeout:45000});await expect.poll(()=>page.getByAltText('当前配方处理水面').evaluate(i=>i.naturalWidth)).toBe(96);
 const routePattern='**/api/projects/'+project.id+'/studio';let delayed=true;
 await page.route(routePattern,async route=>{const response=await route.fetch();const payload=await response.json();log.delayedPayloads.push({requestedAt:Date.now(),jobs:payload.jobs.map(j=>({id:j.id,state:j.state})),successfulBuildId:payload.successfulBuildId});if(delayed)await new Promise(r=>setTimeout(r,1300));await route.fulfill({response}).catch(e=>{log.routeCancelled=(log.routeCancelled??[]).concat(String(e))})});
 const waiter=page.waitForResponse(r=>r.request().method()==='POST'&&r.url().endsWith('/builds'));await page.getByRole('button',{name:'构建当前配套',exact:true}).click();const reply=await waiter;expect(reply.status()).toBe(202);const job=await reply.json();log.job=job.id;
 const item=page.getByRole('list',{name:'构建任务'}).getByRole('listitem').filter({hasText:job.id.slice(0,8)});await expect(item).toBeVisible({timeout:10000});
 let actual;for(let n=0;n<300;n++){actual=await(await fetch(base+`/api/projects/${project.id}/studio`)).json();const j=actual.jobs.find(x=>x.id===job.id);if(['COMPLETED','FAILED'].includes(j.state)){log.backend=j;break}await new Promise(r=>setTimeout(r,100))}expect(log.backend?.state).toBe('COMPLETED');
 await page.waitForTimeout(4500);log.displayedWhileDelayed=await item.innerText();log.successfulWhileDelayed=actual.successfulBuildId;log.completedPayloadCount=log.delayedPayloads.filter(x=>x.jobs.some(j=>j.id===job.id&&j.state==='COMPLETED')).length;log.uiCompleteWhileDelayed=log.displayedWhileDelayed.includes('完成');log.buildButtonDisabled=await page.getByRole('button',{name:'构建当前配套',exact:true}).isDisabled();await page.screenshot({path:out+'/delayed-completed-backend.png',fullPage:true});
 await writeFile(out+'/delayed-observation.json',JSON.stringify(log,null,2)+'\n');delayed=false;await page.unroute(routePattern);await expect(item).toContainText('完成',{timeout:15000});log.displayedAfterNormalLatency=await item.innerText();await page.screenshot({path:out+'/normal-latency-control.png',fullPage:true});
 log.result=log.uiCompleteWhileDelayed?'PASS':'POLL_COMPLETION_STARVATION_REPRODUCED';expect(log.pageErrors).toEqual([]);
}catch(e){log.result='PROBE_FAILURE';log.failure=String(e);await page.screenshot({path:out+'/failure.png',fullPage:true}).catch(()=>{});throw e}finally{await writeFile(out+'/evidence.json',JSON.stringify(log,null,2)+'\n');await context.tracing.stop({path:'.local/step07-review/private/poll-latency-trace.zip'});await browser.close()}
console.log(log.result);
