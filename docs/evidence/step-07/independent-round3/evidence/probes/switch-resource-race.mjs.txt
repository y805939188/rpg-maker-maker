import{chromium,expect}from'@playwright/test';import{readFile,writeFile,mkdir}from'node:fs/promises';
const rec=JSON.parse(await readFile('.local/step07-review/evidence/integrated-recovery.json','utf8')),base='http://127.0.0.1:4383',out='.local/step07-review/evidence/switch-resource-race';await mkdir(out,{recursive:true});const rows=[],browser=await chromium.launch({headless:true});
for(let index=0;index<3;index++){
 const context=await browser.newContext({viewport:{width:1280,height:960}});await context.tracing.start({screenshots:true,snapshots:true,sources:true});const page=await context.newPage(),log={network:[],requests:[],pageErrors:[],console:[]},pending=[];
 page.on('request',r=>{if(r.url().includes('/api/'))log.requests.push({url:r.url(),method:r.method(),time:Date.now()})});page.on('response',r=>{if(r.url().includes('/api/')){const n={url:r.url(),status:r.status(),time:Date.now()};log.network.push(n);if(r.status()>=400)pending.push(r.json().then(b=>n.body=b).catch(()=>{}))}});page.on('pageerror',e=>log.pageErrors.push(e.message));page.on('console',m=>log.console.push({type:m.type(),text:m.text()}));
 try{await page.goto(base);const projects=(await(await fetch(base+'/api/projects')).json()).projects;await page.getByRole('button',{name:projects.find(p=>p.id===rec.legacy.project).name,exact:true}).click();await expect(page.getByText('画布已就绪 · 实际导出图集',{exact:false})).toBeVisible({timeout:45000});log.oldProcessedWidthAtSwitch=await page.getByAltText('当前配方处理水面').evaluate(i=>i.naturalWidth);
  await page.getByRole('button',{name:projects.find(p=>p.id===rec.browserProject).name,exact:true}).click();await expect(page.getByText('画布已就绪 · 实际导出图集',{exact:false})).toBeVisible({timeout:45000});await page.waitForTimeout(2500);log.newProcessedWidth=await page.getByAltText('当前配方处理水面').evaluate(i=>i.naturalWidth);log.selectedBuild=await page.getByLabel('实际构建').inputValue();log.canvasBuild=await page.locator('canvas[aria-label="交付拼接预览"]').getAttribute('data-build-id');log.download=await page.getByRole('link',{name:'下载此构建的标准 MZ 素材包 ZIP'}).getAttribute('href');await page.screenshot({path:out+'/ordinary-switch-'+index+'.png',fullPage:true});await Promise.allSettled(pending);log.result=log.newProcessedWidth===0&&log.network.some(n=>n.status===429&&n.url.includes('/processed?'))?'VALID_PROCESSED_REQUEST_STRANDED':'NORMAL_CONTROL';
 }catch(e){log.failure=String(e);log.result='OTHER_FAILURE';await page.screenshot({path:out+'/failure-'+index+'.png',fullPage:true}).catch(()=>{})}finally{await Promise.allSettled(pending);rows.push(log);await writeFile(out+'/evidence.json',JSON.stringify({networkInjection:false,ordinaryUiOnly:true,rows},null,2)+'\n');await context.tracing.stop({path:'.local/step07-review/private/switch-resource-'+index+'-trace.zip'});await context.close()}
}
await browser.close();console.log(JSON.stringify(rows.map(r=>({old:r.oldProcessedWidthAtSwitch,next:r.newProcessedWidth,result:r.result}))));
