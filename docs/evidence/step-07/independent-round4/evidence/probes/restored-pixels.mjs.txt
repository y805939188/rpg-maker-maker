import {chromium,expect}from'@playwright/test';import{readFile,writeFile,mkdir}from'node:fs/promises';import{createHash}from'node:crypto';
const base='http://127.0.0.1:4385',rec=JSON.parse(await readFile('.local/step07-review/evidence/integrated-recovery.json','utf8')),out='.local/step07-review/evidence/restored-pixels';await mkdir(out,{recursive:true});const sha=b=>createHash('sha256').update(b).digest('hex'),browser=await chromium.launch({headless:true}),page=await browser.newPage({viewport:{width:1280,height:960},acceptDownloads:true});const log={network:[],console:[],pageErrors:[]},pending=[];
page.on('response',r=>{if(/\/files\/|\/profile$/.test(r.url()))pending.push((async()=>{try{const b=await r.body(),n=r.url().endsWith('/profile')?'profile.json':new URL(r.url()).pathname.split('/').pop();await writeFile(out+'/'+n,b);log.network.push({url:r.url(),status:r.status(),sha:sha(b),bytes:b.length})}catch(e){log.network.push({url:r.url(),error:String(e)})}})())});page.on('console',m=>log.console.push({type:m.type(),text:m.text()}));page.on('pageerror',e=>log.pageErrors.push(e.message));
try{await page.goto(base);const projects=(await(await fetch(base+'/api/projects')).json()).projects;await page.getByRole('button',{name:projects.find(p=>p.id===rec.browserProject).name,exact:true}).click();await expect(page.getByText('画布已就绪 · 实际导出图集',{exact:false})).toBeVisible({timeout:30000});const image=page.getByAltText('当前配方处理水面');await expect.poll(()=>image.evaluate(i=>[i.naturalWidth,i.naturalHeight])).toEqual([96,144]);log.processed={project:await image.getAttribute('data-project-id'),recipe:await image.getAttribute('data-recipe-id'),frame:await image.getAttribute('data-source-frame'),sha:await image.getAttribute('data-png-sha256'),pixels:await image.evaluate(i=>{const c=document.createElement('canvas');c.width=96;c.height=144;const ctx=c.getContext('2d');ctx.drawImage(i,0,0);return Array.from(ctx.getImageData(0,0,96,144).data)})};expect(log.processed.project).toBe(rec.browserProject);const response=await fetch(base+`/api/projects/${rec.browserProject}/recipes/${log.processed.recipe}/processed?frame=${log.processed.frame}`);expect(response.status).toBe(200);const png=Buffer.from(await response.arrayBuffer());expect(sha(png)).toBe(log.processed.sha);await writeFile(out+'/processed.png',png);
 const build=await page.getByLabel('实际构建').inputValue();log.build=build;await page.getByLabel('检查场景').selectOption('isolated');const canvas=page.locator('canvas[aria-label="交付拼接预览"]');await expect(canvas).toHaveAttribute('data-scene','isolated',{timeout:30000});await expect(page.getByText('画布已就绪 · 实际导出图集',{exact:false})).toBeVisible();for(let n=0;n<4;n++){await page.getByRole('button',{name:`状态 ${n} · 源帧 ${[0,1,2,1][n]}`,exact:true}).click();const b=Buffer.from(await canvas.evaluate(c=>c.toDataURL().split(',')[1]),'base64');await writeFile(out+'/actual-isolated-'+n+'.png',b)}await page.screenshot({path:out+'/restored-real-image-and-canvas.png',fullPage:true});const wait=page.waitForEvent('download');await page.getByRole('link',{name:'下载此构建的标准 MZ 素材包 ZIP'}).click();await(await wait).saveAs(out+'/current.zip');log.zipSHA=sha(await readFile(out+'/current.zip'));expect(log.pageErrors).toEqual([]);log.result='PASS';}catch(e){log.failure=String(e);log.result='FAIL';await page.screenshot({path:out+'/failure.png',fullPage:true}).catch(()=>{});throw e}finally{await Promise.allSettled(pending);await writeFile(out+'/evidence.json',JSON.stringify(log,null,2)+'\n');await browser.close()}
