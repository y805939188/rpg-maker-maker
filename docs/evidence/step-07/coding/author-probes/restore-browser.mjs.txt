import {chromium} from '@playwright/test';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {createHash} from 'node:crypto';
const record=JSON.parse(await readFile(process.argv[2],'utf8')),root='.local/step07/final/restored-browser';await mkdir(root,{recursive:true});
const browser=await chromium.launch(),page=await browser.newPage({viewport:{width:1200,height:900}}),errors=[];page.on('pageerror',e=>errors.push(e.message));
await page.goto('http://127.0.0.1:4376');const p=record.restoredState.projectId;await page.evaluate(id=>localStorage.setItem('studio-project',id),p);await page.reload();await page.getByRole('heading',{name:'交付预览与构建历史'}).waitFor();await page.getByText('画布已就绪 · 实际导出图集',{exact:false}).waitFor({timeout:30000});const canvas=page.locator('canvas[aria-label="交付拼接预览"]');
const build=await canvas.getAttribute('data-build-id');if(build!==record.rebuilt)throw Error('restored current differs from actual rebuilt Build');await page.screenshot({path:root+'/restored-current.png',fullPage:true});
await page.getByLabel('实际构建').selectOption(record.first);await page.waitForFunction(expected=>document.querySelector('canvas[aria-label="交付拼接预览"]')?.getAttribute('data-build-id')===expected,record.first);await page.getByText('画布已就绪 · 实际导出图集',{exact:false}).waitFor();
const event=page.waitForEvent('download');await page.getByRole('link',{name:'下载此构建的标准 MZ 素材包 ZIP'}).click();const download=await event;await download.saveAs(root+'/restored-history.zip');const bytes=await readFile(root+'/restored-history.zip'),sha=createHash('sha256').update(bytes).digest('hex');if(sha!==record.zipSha)throw Error('restored historical ZIP changed');await page.screenshot({path:root+'/restored-history.png',fullPage:true});
await writeFile(root+'/evidence.json',JSON.stringify({owner:'step07 author real restored browser',projectId:p,restoredBuild:build,historicalBuild:record.first,zipSha:sha,expectedZipSha:record.zipSha,errors,sourceSHA:'4f6a40f3e8be990b4068dafee488249be737f7ad'},null,2)+'\n');await browser.close();if(errors.length)throw Error('page errors');console.log('RESTORED_BROWSER_CURRENT_HISTORY_ZIP_PASS');
