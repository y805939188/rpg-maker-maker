import assert from 'node:assert/strict';
import {resolve} from 'node:path';
import {readFile,writeFile} from 'node:fs/promises';
import {spawn} from 'node:child_process';
import {Persistence} from '@tileset/persistence';
import {PROFILE} from '../../dist/src/profile.js';
import {makeCompilerFixture} from '../../dist/scripts/step04-fixture.js';
import {fragmentsFor,pairOperations} from '../../dist/scripts/step05-fixture.js';
const root=resolve('.local/step06-review');const bound=JSON.parse(await readFile(resolve(root,'evidence/public-probe.json'),'utf8')).checks.find(c=>c.name==='native-bound-publication-control');const store=new Persistence(process.env.DATABASE_URL,resolve(root,'assets'),PROFILE);
async function launch(name,p,stopAt){const path=resolve(root,'private/concurrent-'+name+'.json');await writeFile(path,JSON.stringify({mode:'publish',root:store.storage.root,prepared:{...p,files:Object.fromEntries(Object.entries(p.files).map(([k,v])=>[k,Buffer.from(v).toString('base64')]))},stopAt}));const child=spawn(process.execPath,[resolve(root,'worker.mjs'),path],{env:process.env,stdio:['ignore','pipe','pipe','ipc']});const messages=[];let output='';child.on('message',m=>messages.push(m));child.stdout.on('data',b=>output+=b);child.stderr.on('data',b=>output+=b);const done=new Promise(r=>child.on('close',async(code,signal)=>{await writeFile(path+'.log',output);r({pid:child.pid,code,signal,messages});}));return {child,messages,done};}
async function waitFor(fn){for(let n=0;n<1000;n++){if(await fn())return;await new Promise(r=>setTimeout(r,10));}throw Error('Wait timeout');}
try{
 const fixture=await makeCompilerFixture('art'),fragments=fragmentsFor(fixture),pack=await store.createPack(bound.projectId,'crossprocess-slots-verified');const left=await store.prepareBuild(bound.projectId,pack.id,bound.adoptedId,pairOperations('left',1,fragments),bound.context),right=await store.prepareBuild(bound.projectId,pack.id,bound.adoptedId,pairOperations('right',1,fragments),bound.context);
 const first=await launch('left-stopped',left,'files-durable');await waitFor(()=>first.messages.some(m=>m.point==='files-durable'));const second=await launch('right',right);await waitFor(async()=> (await store.db.$queryRaw`SELECT pid FROM pg_locks WHERE locktype='advisory' AND classid=6007 AND NOT granted`).length>0);
 const sessions=await store.db.$queryRaw`SELECT pid,state,"xact_start" IS NULL AS no_open_transaction FROM pg_stat_activity WHERE datname=current_database() AND pid<>pg_backend_pid()`;assert.ok(sessions.filter(s=>s.state==='idle').every(s=>s.no_open_transaction));assert.ok(sessions.every(s=>s.state!=='idle in transaction'));await writeFile(resolve(root,'private/concurrency-sessions.json'),JSON.stringify(sessions,null,2)+'\n');
 first.child.kill('SIGKILL');const killed=await first.done;assert.equal(killed.signal,'SIGKILL');const complete=await second.done;assert.equal(complete.code,0);assert.equal((await store.db.build.findUniqueOrThrow({where:{id:right.id}})).publication,'ADOPTED');
 const restart=await launch('left-restart',left);assert.equal((await restart.done).code,0);assert.equal((await store.db.build.findUniqueOrThrow({where:{id:left.id}})).publication,'HISTORICAL');assert.equal(await store.db.packVersion.count({where:{packId:pack.id}}),1);
 const current=await store.db.pack.findUniqueOrThrow({where:{id:pack.id}});assert.equal(current.currentBuildId,right.id);const history=await store.db.packVersion.findMany({where:{packId:pack.id},include:{slots:true}});assert.equal(history[0].slots.filter(s=>s.state==='ACTIVE').length,2);
 const record={candidate:bound.context.commit,packId:pack.id,expectedOldVersion:0,left:left.id,right:right.id,blockedSecondObserved:true,sessionsAtFileCheckpoint:sessions,killed,completed:complete,leftAfterRestart:'HISTORICAL',rightPublication:'ADOPTED',canonicalHistory:history.length,currentBuildId:current.currentBuildId,slotBindings:history[0].slots};await writeFile(resolve(root,'evidence/crossprocess-publication.json'),JSON.stringify(record,null,2)+'\n');console.log(JSON.stringify({blockedSecond:true,noIdleTransactions:sessions.every(s=>s.state!=='idle in transaction'),left:'HISTORICAL',right:'ADOPTED'}));
}finally{await store.close();}
